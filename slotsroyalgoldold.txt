    <script>
        // --- GAME CONFIGURATION ---
        const SYMBOLS = ['🍒', '🍋', '🍇', '🍉', '🔔', '💎', '7️⃣', '👑', '🎲'];
        
        // Weighted Randomizer (Rarity: Common -> Rare)
        const WEIGHTS = [
            {sym: '🍒', w: 40}, {sym: '🍋', w: 30}, {sym: '🍇', w: 20}, {sym: '🍉', w: 15},
            {sym: '🔔', w: 10}, {sym: '💎', w: 5}, {sym: '7️⃣', w: 3}, {sym: '👑', w: 1}, {sym: '🎲', w: 4}
        ];

        // Paytable Multipliers
        const PAYOUTS = {
            '🍒': 2, '🍋': 3, '🍇': 5, '🍉': 7, '🔔': 10, '💎': 20, '7️⃣': 30, '👑': 50, '🎲': 0
        };

        // Game State
        let balance = parseFloat(localStorage.getItem('royal_balance')) || 1000.00;
        let bet = 10;
        let jackpot = 12450.00;
        let isSpinning = false;
        let freeSpins = 0;

        // DOM Elements
        const balanceEl = document.getElementById('balance');
        const betEl = document.getElementById('bet-amount');
        const winEl = document.getElementById('last-win');
        const msgEl = document.getElementById('message');
        const gridEl = document.getElementById('grid');
        const spinBtn = document.getElementById('spin-btn');
        const jackpotEl = document.getElementById('jackpot');

        // Initialize Grid (3x3)
        function initGrid() {
            gridEl.innerHTML = '';
            for(let r=0; r<3; r++) {
                for(let c=0; c<3; c++) {
                    let cell = document.createElement('div');
                    cell.className = 'reel-cell';
                    cell.id = `cell-${r}-${c}`;
                    gridEl.appendChild(cell);
                }
            }
        }

        // --- CORE LOGIC ---

        function updateUI() {
            balanceEl.innerText = balance.toFixed(2);
            betEl.innerText = bet;
            winEl.innerText = '$0.00';
            jackpotEl.innerText = Math.floor(jackpot).toLocaleString();
            localStorage.setItem('royal_balance', balance.toString());
        }

        function adjustBet(amount) {
            if(isSpinning) return;
            let newBet = bet + amount;
            if(newBet < 1) newBet = 1;
            if(newBet > balance) newBet = balance;
            bet = newBet;
            updateUI();
        }

        function setBetMax() {
            if(isSpinning) return;
            bet = balance;
            if(bet < 1) bet = 1;
            updateUI();
        }

        function getRandomSymbol() {
            let totalWeight = WEIGHTS.reduce((sum, item) => sum + item.w, 0);
            let random = Math.random() * totalWeight;
            for (let item of WEIGHTS) {
                if (random < item.w) return item.sym;
                random -= item.w;
            }
            return '🍒';
        }

        function spin() {
            if (isSpinning) return;
            if (balance < bet) {
                msgEl.innerText = "Insufficient Funds! 📉";
                msgEl.className = "msg-lose";
                return;
            }

            // Deduct bet
            balance -= bet;
            jackpot += (bet * 0.1); // Jackpot increases with spin
            updateUI();
            
            // Reset Visuals
            msgEl.innerText = "Spinning...";
            msgEl.className = "";
            document.querySelectorAll('.reel-cell').forEach(c => c.classList.remove('win'));
            document.querySelector('.paylines-overlay').classList.remove('payline-active');
            
            isSpinning = true;
            spinBtn.disabled = true;

            // Generate Result (9 cells)
            // We generate the result grid first
            let results = [];
            for(let i=0; i<9; i++) {
                results.push(getRandomSymbol());
            }

            // Animation Loop
            // We create a rapid refresh effect
            const cells = document.querySelectorAll('.reel-cell');
            let speed = 100; // Initial speed
            
            const interval = setInterval(() => {
                cells.forEach(cell => {
                    cell.innerText = getRandomSymbol();
                });
                // Speed up
                if(speed > 20) speed -= 5;
            }, speed);

            // Resolve after 2 seconds
            setTimeout(() => {
                clearInterval(interval);
                isSpinning = false;
                spinBtn.disabled = false;
                
                // Set Final Symbols
                for(let i=0; i<9; i++) {
                    cells[i].innerText = results[i];
                }

                checkWin(results);
            }, 2000);
        }

        function checkWin(results) {
            // results is an array [c0, c1, c2, c3, c4, c5, c6, c7, c8]
            // Grid mapping:
            // Row 0: c0, c1, c2
            // Row 1: c3, c4, c5
            // Row 2: c6, c7, c8
            // Diag 1: c0, c4, c8
            // Diag 2: c2, c4, c6

            let winAmount = 0;
            let winningLines = [];

            // Helper to check line
            const checkLine = (idxs, symbol) => {
                let isWin = true;
                let wilds = 0;
                for (let idx of idxs) {
                    let sym = results[idx];
                    if (sym === '🎲') { isWin = false; break; } // Scatter breaks standard lines usually, but can be separate
                    if (sym === '👑') wilds++;
                    else if (sym !== symbol) { isWin = false; break; }
                }
                return isWin ? { amount: bet * PAYOUTS[symbol], line: idxs } : null;
            };

            // Check 5 Lines
            const lines = [
                { idxs: [0,1,2], name: 'top' }, // Row 0
                { idxs: [3,4,5], name: 'mid' }, // Row 1
                { idxs: [6,7,8], name: 'bot' }, // Row 2
                { idxs: [0,4,8], name: 'diag1' }, // Diag \
                { idxs: [2,4,6], name: 'diag2' }  // Diag /
            ];

            // Check all lines for any match
            for (let line of lines) {
                let sym = results[line.idxs[0]];
                if (sym === '🎲') continue; // Ignore scatter on paylines for standard wins
                
                let res = checkLine(line.idxs, sym);
                if (res) {
                    winAmount += res.amount;
                    winningLines.push(line.name);
                }
            }

            // Check Scatters (Dice)
            let scatterCount = results.filter(s => s === '🎲').length;
            
            if (scatterCount >= 3) {
                let bonusWin = (scatterCount * 5) * bet;
                winAmount += bonusWin;
                msgEl.innerHTML = `BONUS! 🎲 <span style="color:var(--neon-purple)">${freeSpins} Free Spins Active</span>`;
                freeSpins += 3;
                msgEl.className = "msg-bonus";
                triggerBonus();
            } else if (winAmount > 0) {
                // Highlight Wins
                winningLines.forEach(lName => {
                    // Map line name to line IDs for SVG highlighting
                    let ids = lines.find(l => l.name === lName).idxs;
                    document.getElementById(`line-${lName}`).style.stroke = "var(--neon-blue)";
                    ids.forEach(id => document.getElementById(`cell-${id}`).classList.add('win'));
                });
                
                balance += winAmount;
                msgEl.innerHTML = `WIN! +$${winAmount.toFixed(2)} 💰`;
                msgEl.className = "msg-win";
            } else {
                msgEl.innerText = "Try again!";
                msgEl.className = "msg-lose";
            }
            
            updateUI();
        }

        function triggerBonus() {
            // Simple visual feedback for bonus
            let count = 0;
            const bonusAnim = setInterval(() => {
                count++;
                msgEl.innerText = "BONUS ROUND! 🚀";
                document.body.style.boxShadow = "inset 0 0 50px var(--neon-purple)";
                if(count > 10) {
                    clearInterval(bonusAnim);
                    document.body.style.boxShadow = "none";
                }
            }, 100);
        }

        function togglePaytable() {
            let p = document.getElementById('paytable');
            p.style.display = p.style.display === 'flex' ? 'none' : 'flex';
        }

        // Initialize
        initGrid();
        updateUI();

    </script>
